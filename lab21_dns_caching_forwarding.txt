RHCE RH254 HANDS-ON LAB: DNS CACHING AND FORWARDING
===================================================

LAB OBJECTIVE:
Configure DNS caching and forwarding to improve DNS performance and provide efficient name resolution

PREREQUISITES:
- BIND DNS server installed and configured
- Root access to RHEL system
- Understanding of DNS caching concepts

LAB SCENARIO:
Configure DNS server as a caching nameserver with conditional forwarding for different domains and networks.

EQUIPMENT NEEDED:
- RHEL system with BIND installed
- Multiple DNS servers for forwarding
- Client systems for testing

LAB TASKS:

PART A: CONFIGURE DNS CACHING
------------------------------

dnf install -y bind bind-utils
systemctl enable --now named

1. Configure basic caching settings:
   # vim /etc/named.conf
   
   options {
    listen-on port 53 { 127.0.0.1; };
    listen-on-v6 { ::1; };

    directory "/var/named";

    recursion yes;

    allow-recursion { localhost; };
    allow-query-cache { localhost; };

    dnssec-validation yes;

    max-cache-size 512M;
    max-cache-ttl 86400;
    max-ncache-ttl 3600;

    minimal-responses yes;
};

        file "/var/log/named/cache.log" versions 3 size 5m;
        severity info;
        print-time yes;
        print-category yes;
    };

    category resolver { cache_log; };
};

PART E: CONFIGURE CACHE MANAGEMENT
-----------------------------------

1. Create cache management script:
   # vim /usr/local/bin/dns-cache-manager.sh
   
   #!/bin/bash
# DNS cache management script (RHEL 9)

case "$1" in
    flush)
        echo "Flushing DNS cache..."
        rndc flush
        echo "Cache flushed."
        ;;
    stats)
        echo "Generating DNS cache statistics..."
        rndc stats
        if [ -f /var/named/data/named_stats.txt ]; then
            tail -20 /var/named/data/named_stats.txt
        else
            echo "Stats file not found."
        fi
        ;;
    dump)
        echo "Dumping DNS cache..."
        rndc dumpdb -cache
        echo "Cache dumped to /var/named/data/cache_dump.db"
        ;;
    size)
        echo "DNS cache status:"
        rndc status | grep -i cache
        ;;
    *)
        echo "Usage: $0 {flush|stats|dump|size}"
        exit 1
        ;;
esac

   # chmod +x /usr/local/bin/dns-cache-manager.sh

2. Configure automatic cache cleaning:
   # vim /usr/local/bin/dns-cache-clean.sh
   
   #!/bin/bash
# Automatic DNS cache cleaning script (RHEL 9)

LOG_FILE="/var/log/dns-cache-clean.log"

CACHE_LINE=$(rndc status | grep -i "cache size")
CACHE_SIZE=$(echo "$CACHE_LINE" | awk '{print $NF}')

if [ -z "$CACHE_SIZE" ]; then
    echo "$(date): Unable to determine cache size" >> "$LOG_FILE"
    exit 0
fi

# Threshold: 400 MB
if [ "$CACHE_SIZE" -gt 400000000 ]; then
    echo "$(date): Cache size ${CACHE_SIZE} bytes exceeds limit, flushing" >> "$LOG_FILE"
    rndc flush
    echo "$(date): Cache flushed" >> "$LOG_FILE"
else
    echo "$(date): Cache size ${CACHE_SIZE} bytes OK" >> "$LOG_FILE"
fi

   # chmod +x /usr/local/bin/dns-cache-clean.sh

PART G: MONITOR CACHING PERFORMANCE
------------------------------------

1. Create cache monitoring script:
   # vim /usr/local/bin/cache-monitor.sh
   
   #!/bin/bash
# DNS cache monitoring script (RHEL 9)

STATS_FILE="/var/named/data/named_stats.txt"

echo "=== DNS Cache Performance Report ==="
echo "Date: $(date)"
echo

# Generate fresh stats
rndc stats >/dev/null 2>&1

if [ -f "$STATS_FILE" ]; then
    echo "Recent Cache Statistics:"
    grep -E "cache hits|cache misses|queries received" "$STATS_FILE" | tail -10
    echo

    HITS=$(grep "cache hits" "$STATS_FILE" | tail -1 | awk '{print $1}')
    MISSES=$(grep "cache misses" "$STATS_FILE" | tail -1 | awk '{print $1}')

    if [[ -n "$HITS" && -n "$MISSES" ]]; then
        TOTAL=$((HITS + MISSES))
        if [ "$TOTAL" -gt 0 ]; then
            RATIO=$((HITS * 100 / TOTAL))
            echo "Cache Hit Ratio: ${RATIO}%"
        fi
    fi
else
    echo "Statistics file not found (insufficient traffic)."
fi

echo
echo "Current Resolver Status:"
rndc status | grep -E "recursive clients|cache size"

   # chmod +x /usr/local/bin/cache-monitor.sh

1. Create cache troubleshooting script:
   # vim /usr/local/bin/cache-troubleshoot.sh
   
   #!/bin/bash
# DNS cache troubleshooting script (RHEL 9)

echo "=============================="
echo " DNS CACHE TROUBLESHOOT REPORT"
echo "=============================="
echo "Date: $(date)"
echo

echo "1) BIND Service Status:"
systemctl is-active named && systemctl is-enabled named
echo

echo "2) Configuration Check:"
if named-checkconf >/dev/null 2>&1; then
    echo "named.conf: OK"
else
    echo "named.conf: ERROR"
fi
echo

echo "3) Cache Status:"
rndc status | grep -E "cache size|recursive clients"
echo

echo "4) Forwarder Reachability:"
for fw in 8.8.8.8 8.8.4.4; do
    if dig @$fw google.com +time=1 +tries=1 +short >/dev/null; then
        echo "Forwarder $fw: OK"
    else
        echo "Forwarder $fw: UNREACHABLE"
    fi
done
echo

echo "5) RPZ Validation:"
dig @127.0.0.1 malware.example.com +short >/dev/null
if [ $? -ne 0 ]; then
    echo "RPZ blocking: ACTIVE"
else
    echo "RPZ blocking: FAILED"
fi
echo

echo "6) Recent named log entries:"
journalctl -u named -n 10 --no-pager
