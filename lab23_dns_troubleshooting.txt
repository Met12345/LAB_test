dig @$DNS_SERVER $DOMAIN ANY
echo

echo "4. Resolution trace:"
dig @$DNS_SERVER $DOMAIN +trace
echo

IP=$(dig @$DNS_SERVER $DOMAIN A +short | head -1)
if [ -n "$IP" ]; then
    echo "5. Reverse DNS for $IP:"
    dig @$DNS_SERVER -x $IP +short
fi

1. Create comprehensive troubleshooting script:
   # vim /usr/local/bin/dns-troubleshoot.sh
   
   #!/bin/bash
# DNS Troubleshooting Script (RHEL 9 SAFE)

DOMAIN="$1"
DNS_SERVER="${2:-localhost}"

echo "=== DNS TROUBLESHOOTING REPORT ==="
echo "Domain     : ${DOMAIN:-N/A}"
echo "DNS Server : $DNS_SERVER"
echo "Date       : $(date)"
echo "================================="
echo

# 1. DNS service status
echo "1. DNS Service Status:"
if systemctl is-active named >/dev/null 2>&1; then
    echo "  named service: RUNNING"
else
    echo "  named service: NOT RUNNING"
fi
echo

# 2. named.conf syntax check
echo "2. Configuration Syntax Check:"
if named-checkconf >/dev/null 2>&1; then
    echo "  named.conf syntax: OK"
else
    echo "  named.conf syntax: ERROR"
    named-checkconf
fi
echo

# 3. Zone file validation
echo "3. Zone File Validation:"
for zone_file in /var/named/*.zone; do
    [ -f "$zone_file" ] || continue
    zone_name=$(basename "$zone_file" .zone)
    if named-checkzone "$zone_name" "$zone_file" >/dev/null 2>&1; then
        echo "  $zone_name: OK"
    else
        echo "  $zone_name: ERROR"
        named-checkzone "$zone_name" "$zone_file" | head -3
    fi
done
echo

# 4. Port and network checks
echo "4. Network Checks:"
if ss -lupn | grep -q ":53"; then
    echo "  DNS port 53 listening: YES"
else
    echo "  DNS port 53 listening: NO"
fi

if ping -c1 -W2 8.8.8.8 >/dev/null 2>&1; then
    echo "  External network reachability: OK"
else
    echo "  External network reachability: FAILED"
fi
echo

# 5. DNS resolution tests
if [ -n "$DOMAIN" ]; then
    echo "5. DNS Resolution Test:"

    local_result=$(dig @"$DNS_SERVER" "$DOMAIN" +time=2 +tries=1 +short)
    if [ -n "$local_result" ]; then
        echo "  Local resolution: OK -> $local_result"
    else
        echo "  Local resolution: FAILED"
        dig @"$DNS_SERVER" "$DOMAIN" | grep "status:"
    fi

    external_result=$(dig @8.8.8.8 "$DOMAIN" +time=2 +tries=1 +short)
    if [ -n "$external_result" ]; then
        echo "  External resolution: OK"
    else
        echo "  External resolution: FAILED"
    fi
fi
echo

# 6. Recent DNS errors (journalctl)
echo "6. Recent named errors (last 15 min):"
journalctl -u named --since "15 minutes ago" | grep -i error || echo "  No recent errors"
echo

   
   # chmod +x /usr/local/bin/dns-troubleshoot.sh
echo $PATH | tr ':' '\n' | grep /usr/local/bin
echo 'export PATH=$PATH:/usr/local/bin' >> ~/.bashrc
source ~/.bashrc

2. Create performance troubleshooting script:
   # vim /usr/local/bin/dns-performance.sh
   
   #!/bin/bash
# DNS Performance Troubleshooting Script (RHEL 9 SAFE)

DOMAIN="$1"
DNS_SERVER="${2:-localhost}"
ITERATIONS=10

if [ -z "$DOMAIN" ]; then
    echo "Usage: dns-performance.sh <domain> [dns_server]"
    exit 1
fi

echo "=== DNS PERFORMANCE ANALYSIS ==="
echo "Domain     : $DOMAIN"
echo "DNS Server : $DNS_SERVER"
echo "Iterations : $ITERATIONS"
echo

# 1. Query response time test
echo "1. Query Response Times:"
TOTAL_TIME=0

for i in $(seq 1 $ITERATIONS); do
    START_TIME=$(date +%s%N)
    dig @"$DNS_SERVER" "$DOMAIN" +time=2 +tries=1 +short >/dev/null 2>&1
    END_TIME=$(date +%s%N)

    QUERY_TIME=$(( (END_TIME - START_TIME) / 1000000 ))
    echo "  Query $i: ${QUERY_TIME} ms"

    TOTAL_TIME=$((TOTAL_TIME + QUERY_TIME))
done

AVERAGE_TIME=$((TOTAL_TIME / ITERATIONS))
echo "  Average response time: ${AVERAGE_TIME} ms"
echo

# 2. Cache performance test
echo "2. Cache Performance Test:"
echo "  First query (possible cache miss):"
time dig @"$DNS_SERVER" "$DOMAIN" +time=2 +tries=1 >/dev/null 2>&1

echo "  Second query (expected cache hit):"
time dig @"$DNS_SERVER" "$DOMAIN" +time=2 +tries=1 >/dev/null 2>&1
echo

# 3. DNS server statistics
echo "3. DNS Server Statistics:"
if command -v rndc >/dev/null 2>&1; then
    rndc stats >/dev/null 2>&1
    if [ -f /var/named/data/named_stats.txt ]; then
        echo "  Recent statistics:"
        tail -10 /var/named/data/named_stats.txt
    else
        echo "  Statistics file not found"
    fi
else
    echo "  rndc command not available"
fi

echo
echo "=== PERFORMANCE TEST COMPLETE ==="

   
   # chmod +x /usr/local/bin/dns-performance.sh
bash -n /usr/local/bin/dns-performance.sh
dns-performance.sh google.com

1. Create NXDOMAIN troubleshooting script:
   # vim /usr/local/bin/diagnose-nxdomain.sh
   
   #!/bin/bash
# NXDOMAIN Troubleshooting Script (RHEL 9 SAFE)

DOMAIN="$1"
DNS_SERVER="${2:-localhost}"

if [ -z "$DOMAIN" ]; then
    echo "Usage: diagnose-nxdomain.sh <domain> [dns_server]"
    exit 1
fi

echo "=== NXDOMAIN DIAGNOSIS REPORT ==="
echo "Domain     : $DOMAIN"
echo "DNS Server : $DNS_SERVER"
echo "Date       : $(date)"
echo "================================"
echo

# 1. Authoritative server check
echo "1. Checking authoritative name servers:"

PARENT_DOMAIN=$(echo "$DOMAIN" | cut -d. -f2-)

NS_SERVERS=$(dig "$PARENT_DOMAIN" NS +short)

if [ -z "$NS_SERVERS" ]; then
    echo "  No authoritative name servers found"
else
    for ns in $NS_SERVERS; do
        echo "  Querying $ns:"
        result=$(dig @"$ns" "$DOMAIN" A +time=2 +tries=1 +short)
        if [ -n "$result" ]; then
            echo "    A record found -> $result"
        else
            echo "    NXDOMAIN or no response"
        fi
    done
fi
echo

# 2. Local zone file check
echo "2. Checking local zone files:"

FOUND_LOCAL="no"
for zone_file in /var/named/*.zone; do
    [ -f "$zone_file" ] || continue
    if grep -q "$DOMAIN" "$zone_file"; then
        echo "  Found in $zone_file"
        FOUND_LOCAL="yes"
    fi
done

if [ "$FOUND_LOCAL" = "no" ]; then
    echo "  Domain not found in local zones"
fi
echo

# 3. Forwarder configuration check
echo "3. Checking forwarders configuration:"
if grep -q "forwarders" /etc/named.conf; then
    grep -A 5 "forwarders" /etc/named.conf
else
    echo "  No forwarders configured"
fi
echo

# 4. Record type testing
echo "4. Testing record types:"

for type in A AAAA CNAME MX NS; do
    record=$(dig @"$DNS_SERVER" "$DOMAIN" "$type" +time=2 +tries=1 +short)
    if [ -n "$record" ]; then
        echo "  $type record: $record"
    else
        echo "  $type record: NXDOMAIN / no data"
    fi
done

echo
echo "=== NXDOMAIN DIAGNOSIS COMPLETE ==="

   
   # chmod +x /usr/local/bin/diagnose-nxdomain.sh
bash -n /usr/local/bin/diagnose-nxdomain.sh
diagnose-nxdomain.sh nonexistent.example.com

   #!/bin/bash

    echo "  Ping to $DNS_SERVER: OK"

    echo "  Ping to $DNS_SERVER: FAILED"

if command -v nc >/dev/null 2>&1; then
    if nc -z -w 5 "$DNS_SERVER" 53 >/dev/null 2>&1; then
        echo "  Port 53 connectivity: OK"
    else
        echo "  Port 53 connectivity: FAILED"
    fi

    echo "  nc command not available"

# 2. Firewall status (RHEL 9)

        echo "  DNS service allowed in firewalld"

        echo "  DNS service NOT allowed in firewalld"
    fi
else
    echo "  firewalld not running"
fi

echo

# 3. DNS timeout testing
echo "3. DNS Timeout Tests:"

for t in 1 5 10 30; do
    echo "  Testing with timeout ${t}s:"
    dig @"$DNS_SERVER" "$DOMAIN" +time=$t +tries=1 >/dev/null 2>&1
    if [ $? -eq 0 ]; then
        echo "    SUCCESS"
        break
    else
        echo "    TIMEOUT"
    fi

echo

# 4. Server load
echo "4. Server Load & Resources:"

echo "  Load average:"
uptime

echo "  named process count:"
ps -C named --no-headers | wc -l

echo "=== TIMEOUT DIAGNOSIS COMPLETE ==="

2. Create real-time DNS monitoring:
   # vim /usr/local/bin/monitor-dns-realtime.sh
   
   #!/bin/bash
# Real-time DNS Monitoring Script (RHEL 9 SAFE)

echo "=== REAL-TIME DNS MONITORING ==="
echo "Service : named"
echo "Press   : Ctrl+C to stop"
echo "================================"
echo

# Follow named logs in real time using journalctl
journalctl -u named -f | while read -r line; do

    # Extract timestamp (first 3 fields)
    timestamp=$(echo "$line" | awk '{print $1, $2, $3}')

    # Full message
    message="$line"

    # Highlight based on message type
    if echo "$message" | grep -qi "error\|failed"; then
        echo -e "\033[31m[$timestamp] ERROR   : $message\033[0m"

    elif echo "$message" | grep -qi "warning\|denied\|refused"; then
        echo -e "\033[33m[$timestamp] WARNING : $message\033[0m"

    elif echo "$message" | grep -qi "started\|listening\|loaded"; then
        echo -e "\033[32m[$timestamp] INFO    : $message\033[0m"

    else
        echo "[$timestamp] $message"
    fi

done

#!/bin/bash
# DNS Connectivity Tester Script (RHEL 9 SAFE)

DNS_SERVERS=(
    "8.8.8.8:Google"
    "8.8.4.4:Google"
    "1.1.1.1:Cloudflare"
    "1.0.0.1:Cloudflare"
    "208.67.222.222:OpenDNS"
    "208.67.220.220:OpenDNS"
    "localhost:Local"
)

TEST_DOMAIN="${1:-google.com}"

echo "=== DNS CONNECTIVITY TEST ==="
echo "Test domain : $TEST_DOMAIN"
echo "================================"
echo

for entry in "${DNS_SERVERS[@]}"; do
    server="${entry%%:*}"
    provider="${entry##*:}"

    printf "%-15s %-12s " "$server" "($provider)"

    # UDP test
    dig @"$server" "$TEST_DOMAIN" +time=2 +tries=1 +short >/dev/null 2>&1
    if [ $? -eq 0 ]; then
        printf "UDP:OK   "
    else
        printf "UDP:FAIL "
    fi

    # TCP test
    dig @"$server" "$TEST_DOMAIN" +tcp +time=2 +tries=1 +short >/dev/null 2>&1
    if [ $? -eq 0 ]; then
        printf "TCP:OK"
    else
        printf "TCP:FAIL"
    fi

    echo
done

echo
echo "=== CONNECTIVITY TEST COMPLETE ==="

1. Create DNS health check script:
   # vim /usr/local/bin/dns-health-check.sh
   
   #!/bin/bash
# Automated DNS Health Check Script (RHEL 9 SAFE)

HEALTH_SCORE=100
ISSUES=()

echo "=== DNS HEALTH CHECK REPORT ==="
echo "Date : $(date)"
echo "==============================="
echo

# 1. DNS service status
if systemctl is-active named >/dev/null 2>&1; then
    :
else
    HEALTH_SCORE=$((HEALTH_SCORE - 30))
    ISSUES+=("DNS service is not running")
fi

# 2. named.conf syntax
if named-checkconf >/dev/null 2>&1; then
    :
else
    HEALTH_SCORE=$((HEALTH_SCORE - 25))
    ISSUES+=("named.conf has syntax errors")
fi

# 3. Zone file validation
ZONE_ERRORS=0
for zone_file in /var/named/*.zone; do
    [ -f "$zone_file" ] || continue
    zone_name=$(basename "$zone_file" .zone)
    if named-checkzone "$zone_name" "$zone_file" >/dev/null 2>&1; then
        :
    else
        ZONE_ERRORS=$((ZONE_ERRORS + 1))
    fi
done

if [ "$ZONE_ERRORS" -gt 0 ]; then
    HEALTH_SCORE=$((HEALTH_SCORE - (ZONE_ERRORS * 10)))
    ISSUES+=("$ZONE_ERRORS zone file(s) have errors")
fi

# 4. External DNS connectivity
if dig @8.8.8.8 google.com +time=2 +tries=1 +short >/dev/null 2>&1; then
    :
else
    HEALTH_SCORE=$((HEALTH_SCORE - 15))
    ISSUES+=("External DNS connectivity failed")
fi

# 5. Local DNS resolution
if dig @localhost google.com +time=2 +tries=1 +short >/dev/null 2>&1; then
    :
else
    HEALTH_SCORE=$((HEALTH_SCORE - 20))
    ISSUES+=("Local DNS resolution failed")
fi

# 6. Recent errors from logs (journalctl)
ERROR_COUNT=$(journalctl -u named --since "1 hour ago" | grep -Ei "error|failed" | wc -l)

if [ "$ERROR_COUNT" -gt 10 ]; then
    HEALTH_SCORE=$((HEALTH_SCORE - 10))
    ISSUES+=("High error count in logs: $ERROR_COUNT")
fi

# Report
echo "Health Score: $HEALTH_SCORE / 100"
echo

if [ "${#ISSUES[@]}" -eq 0 ]; then
    echo "✓ All checks passed – DNS service is healthy"
else
    echo "Issues detected:"
    for issue in "${ISSUES[@]}"; do
        echo "  ✗ $issue"
    done
fi

echo
echo "Recommendations:"
if [ "$HEALTH_SCORE" -lt 70 ]; then
    echo "  - Immediate attention required"
    echo "  - Run: dns-troubleshoot.sh"
elif [ "$HEALTH_SCORE" -lt 90 ]; then
    echo "  - Minor issues detected"
    echo "  - Review configuration and logs"
else
    echo "  - DNS service operating normally"
    echo "  - Continue monitoring"
fi

echo
echo "=== HEALTH CHECK COMPLETE ==="

   
   # chmod +x /usr/local/bin/dns-health-check.sh
bash -n /usr/local/bin/dns-health-check.sh
dns-health-check.sh
