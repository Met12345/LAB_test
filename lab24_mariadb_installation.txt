RHCE RH254 HANDS-ON LAB: MARIADB/MYSQL INSTALLATION AND CONFIGURATION
====================================================================

LAB OBJECTIVE:
Install and configure MariaDB/MySQL database server with basic settings and initial database setup

PREREQUISITES:
- RHEL 7/8 system with root access
- Network connectivity
- Understanding of database concepts

LAB SCENARIO:
Install MariaDB database server, configure basic settings, create databases and users for application use.

EQUIPMENT NEEDED:
- RHEL system (db.example.com: 192.168.1.10)
- Sufficient disk space for database storage
- Client systems for testing connectivity

LAB TASKS:

PART A: INSTALL MARIADB/MYSQL
------------------------------

1. Install MariaDB server and client:
   # yum install mariadb-server mariadb -y
   # systemctl enable mariadb

2. Verify installation:
   # rpm -qa | grep mariadb
   # mysql --version
   # mysqld --version

3. Check default configuration:
   # ls -la /etc/my.cnf*
   # ls -la /var/lib/mysql/

4. Start MariaDB service:
   # systemctl start mariadb
   # systemctl status mariadb

PART C: CONFIGURE MARIADB SETTINGS
-----------------------------------

1. Configure main MariaDB settings:
   # vim /etc/my.cnf.d/custom-server.cnf
   
[mysqld]

# Network
bind-address = 0.0.0.0
port = 3306

# Paths
datadir = /var/lib/mysql
socket = /var/lib/mysql/mysql.sock

# Logging
log-error = /var/log/mariadb/mariadb.log
slow_query_log = ON
slow_query_log_file = /var/log/mariadb/slow.log
long_query_time = 2

# Character set
character-set-server = utf8mb4
collation-server = utf8mb4_unicode_ci

# InnoDB
innodb_buffer_pool_size = 256M
innodb_log_file_size = 64M
innodb_file_per_table = ON

# Connections
max_connections = 100
max_user_connections = 50

~DO NOT ADD

query_cache
mysqld_safe
random tuning copied from blogs

2. Create log directory:
   # mkdir -p /var/log/mariadb
   # chown mysql:mysql /var/log/mariadb
   # chmod 755 /var/log/mariadb

3. Configure client settings:
   vim /etc/my.cnf.d/client-charset.cnf
   
   [client]
   default-character-set = utf8mb4
   
   [mysql]
   default-character-set = utf8mb4

ls -l /etc/my.cnf.d/
cat /etc/my.cnf.d/custom-server.cnf
cat /etc/my.cnf.d/client-charset.cnf

--------------------------------------

1. Enable performance schema:
mysql -u root -p -e "SHOW VARIABLES LIKE 'performance_schema';"

   # vim /etc/my.cnf.d/performance.cnf

   
   [mysqld]
   # Performance monitoring
   performance_schema = ON
   performance_schema_max_table_instances = 400
   performance_schema_max_table_handles = 4000

2. Create monitoring user:
   # mysql -u root -p
   
   MariaDB> CREATE USER 'monitor'@'localhost' IDENTIFIED BY 'MonitorPass123!';
   MariaDB> GRANT PROCESS, REPLICATION CLIENT ON *.* TO 'monitor'@'localhost';
   MariaDB> GRANT SELECT ON performance_schema.* TO 'monitor'@'localhost';
   MariaDB> FLUSH PRIVILEGES;
SHOW GRANTS FOR 'monitor'@'localhost';
   MariaDB> EXIT;

3. Create database monitoring script:
   # vim /usr/local/bin/db-monitor.sh
   
#!/bin/bash
# MariaDB Monitoring Script (RHEL 9 / RHCE / Production Safe)

DB_USER="monitor"
DB_PASS="MonitorPass123!"

echo "=== MariaDB Status Report ==="
date
echo

echo "Active Connections:"
mysql -u"$DB_USER" -p"$DB_PASS" -e "SHOW STATUS LIKE 'Threads_connected';"
echo

echo "Database Sizes (MB):"
mysql -u"$DB_USER" -p"$DB_PASS" <<EOF
SELECT
  table_schema AS db_name,
  ROUND(SUM(data_length + index_length)/1024/1024,2) AS size_mb
FROM information_schema.tables
GROUP BY table_schema;
EOF
echo

echo "Slow Queries:"
mysql -u"$DB_USER" -p"$DB_PASS" -e "SHOW STATUS LIKE 'Slow_queries';"
echo

echo "InnoDB Buffer Pool:"
mysql -u"$DB_USER" -p"$DB_PASS" <<EOF
SELECT VARIABLE_NAME, VARIABLE_VALUE
FROM performance_schema.global_status
WHERE VARIABLE_NAME IN
('Innodb_buffer_pool_pages_total','Innodb_buffer_pool_pages_free');
EOF
   chmod 750 /usr/local/bin/db-monitor.sh

PART H: CONFIGURE BACKUP STRATEGY
----------------------------------

1. Create backup directory:
   # mkdir -p /var/backups/mysql
   # chown mysql:mysql /var/backups/mysql
   # chmod 750 /var/backups/mysql
ls -ld /var/backups/mysql

2. Create backup script:
   # vim /usr/local/bin/mysql-backup.sh
   
#!/bin/bash
# MariaDB Full Backup Script (RHEL 9 / RHCE / Production Safe)

BACKUP_DIR="/var/backups/mysql"
DB_USER="backup_user"
DB_PASS="BackupPass123!"
DATE="$(date +%Y%m%d_%H%M%S)"
TMP_FILE="/tmp/mysql_backup_${DATE}.sql"
FINAL_FILE="${BACKUP_DIR}/full_backup_${DATE}.sql.gz"

echo "Starting MariaDB backup at $(date)"

mysqldump -u"$DB_USER" -p"$DB_PASS" --all-databases --single-transaction --routines --triggers --lock-tables=false > "$TMP_FILE"

if [ $? -ne 0 ]; then
    echo "ERROR: mysqldump failed"
    rm -f "$TMP_FILE"
    exit 1
fi

gzip "$TMP_FILE"

mv "${TMP_FILE}.gz" "$FINAL_FILE"

echo "Backup completed successfully"
echo "Backup file: $FINAL_FILE"

find "$BACKUP_DIR" -type f -name "full_backup_*.sql.gz" -mtime +7 -delete

   chmod 750 /usr/local/bin/mysql-backup.sh
chown root:root /usr/local/bin/mysql-backup.sh
   # chmod +x /usr/local/bin/mysql-backup.sh

2. Create performance tuning script:
   # vim /usr/local/bin/mysql-tuning.sh
   
   #!/bin/bash
# MySQL performance analysis script
# Safe for production (read-only)

DB_USER="root"

echo "=== MySQL Performance Analysis ==="
echo "Date: $(date)"
echo

echo "Database Size (webapp_prod):"
mysql -u "$DB_USER" -p -e "
SELECT
  table_schema AS database_name,
  ROUND(SUM(data_length + index_length)/1024/1024, 2) AS size_mb
FROM information_schema.tables
WHERE table_schema='webapp_prod'
GROUP BY table_schema;
"
echo

echo "Query Cache Statistics:"
mysql -u "$DB_USER" -p -e "
SHOW STATUS
WHERE Variable_name IN (
  'Qcache_queries_in_cache',
  'Qcache_inserts',
  'Qcache_hits',
  'Qcache_lowmem_prunes'
);
"
echo

echo "Connection Statistics:"
mysql -u "$DB_USER" -p -e "
SHOW STATUS
WHERE Variable_name IN (
  'Connections',
  'Max_used_connections',
  'Threads_connected'
);
"

   #!/bin/bash
# Slow query analysis script
# Safe for production (read-only)

SLOW_LOG="/var/log/mariadb/slow.log"

echo "=== Slow Query Analysis ==="
echo "Date: $(date)"
echo

if [ ! -f "$SLOW_LOG" ]; then
  echo "Slow query log not found"

echo "Total slow queries:"

echo
