RHCE RH254 HANDS-ON LAB: CUPS PRINT SERVER SETUP
===============================================

LAB OBJECTIVE:
Configure CUPS (Common Unix Printing System) print server for network printing, including printer installation, sharing, access control, and print queue management

PREREQUISITES:
- RHEL 8/9 system with root access
- Network printer or printer simulator
- Understanding of printing concepts
- Basic knowledge of web interface administration

LAB SCENARIO:
Set up a centralized print server using CUPS to manage network printers, configure print queues, implement access controls, and provide printing services to network clients.

EQUIPMENT NEEDED:
- RHEL system (server: 20.192.255.178)
- Client systems for testing (20.192.213.99)
- Network printer (20.192.250.238) or USB printer

LAB TASKS:

PART A: INSTALL AND CONFIGURE CUPS
-----------------------------------

1. Install CUPS packages:
   # dnf install cups cups-client cups-filters -y
   # systemctl enable cups
   # systemctl start cups

2. Configure CUPS daemon:
   # vim /etc/cups/cupsd.conf
   
   # Listen on all interfaces
   Listen 631
   Listen /var/run/cups/cups.sock
   
   # Share printers on the local network
   Browsing On
   BrowseLocalProtocols dnssd
   
   # Default authentication type
   DefaultAuthType Basic
   
   # Web interface and administration
   WebInterface Yes
   
   # Location directives
   <Location />
     Allow localhost
     Allow 20.192.0.0/16
   </Location>
   
   <Location /admin>
     Allow localhost
     Allow 20.192.0.0/16
   </Location>
   
   <Location /admin/conf>
     AuthType Default
     Require user @SYSTEM
     Allow localhost

1. Create print job management script:
   # vim /usr/local/bin/manage-print-jobs.sh
   
   #!/bin/bash
   
   case $1 in
       "list")
           echo "Active print jobs:"
           lpstat -o
           ;;
       "cancel")
           if [ -n "$2" ]; then
               cancel $2
               echo "Job $2 cancelled"
           else
               echo "Usage: $0 cancel <job-id>"
           fi
           ;;
       "hold")
           if [ -n "$2" ]; then
               lp -i $2 -H hold
               echo "Job $2 held"
           fi
           ;;
       "release")
           if [ -n "$2" ]; then
               lp -i $2 -H resume
               echo "Job $2 released"
           fi
           ;;
       "purge")
           cancel -a
           echo "All jobs cancelled"
           ;;
       *)
           echo "Usage: $0 {list|cancel|hold|release|purge} [job-id]"
           ;;
   esac

   echo "=== Printer Maintenance - $(date) ==="

           systemctl reload cups
       fi
   fi
   
   echo "Maintenance completed"

#!/bin/bash

LOGFILE="/var/log/print-monitor.log"
CUPS_ERROR_LOG="/var/log/cups/error_log"

# Ensure log file exists and is writable
touch "$LOGFILE" 2>/dev/null || exit 1

while true; do
    echo "=== Print Queue Status - $(date) ===" >> "$LOGFILE"

    # Check printer status
    if command -v lpstat >/dev/null 2>&1; then
        lpstat -p >> "$LOGFILE" 2>&1
    else
        echo "lpstat command not found" >> "$LOGFILE"
    fi
    echo >> "$LOGFILE"

    # Check print jobs
    if command -v lpstat >/dev/null 2>&1; then
        lpstat -o >> "$LOGFILE" 2>&1
    else
        echo "lpstat command not found" >> "$LOGFILE"
    fi
    echo >> "$LOGFILE"

    # Check for errors
    if [ -r "$CUPS_ERROR_LOG" ]; then
        tail -10 "$CUPS_ERROR_LOG" >> "$LOGFILE"
    else
        echo "CUPS error log not found or not readable" >> "$LOGFILE"
    fi

    echo "---" >> "$LOGFILE"

    sleep 300
done

#!/bin/bash

echo "=== Printer Maintenance - $(date) ==="

# Clean print queues
echo "Cleaning completed jobs..."
if command -v cancel >/dev/null 2>&1; then
    cancel -a -x 2>/dev/null
fi

# Restart CUPS if needed
if ! systemctl is-active cups >/dev/null 2>&1; then
    echo "Restarting CUPS service..."
    systemctl restart cups 2>/dev/null
fi

# Check disk space for spool directory
if [ -d /var/spool/cups ]; then
    SPOOL_USAGE=$(df /var/spool/cups 2>/dev/null | tail -1 | awk '{print $5}' | cut -d'%' -f1)
    if [ -n "$SPOOL_USAGE" ] && [ "$SPOOL_USAGE" -gt 80 ]; then
        echo "WARNING: Print spool directory is ${SPOOL_USAGE}% full"
        find /var/spool/cups -name "c*" -mtime +7 -delete 2>/dev/null
        find /var/spool/cups -name "d*" -mtime +7 -delete 2>/dev/null
    fi
fi

# Rotate logs
if [ -f /var/log/cups/access_log ]; then
    LOG_SIZE=$(stat -c%s /var/log/cups/access_log 2>/dev/null)
    if [ -n "$LOG_SIZE" ] && [ "$LOG_SIZE" -gt 10485760 ]; then
        mv /var/log/cups/access_log /var/log/cups/access_log.old 2>/dev/null
        systemctl reload cups 2>/dev/null
    fi
fi

echo "Maintenance completed"
