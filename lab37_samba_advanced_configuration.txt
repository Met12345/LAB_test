RHCE RH254 HANDS-ON LAB: SAMBA FILE SHARING ADVANCED CONFIGURATION
================================================================

LAB OBJECTIVE:
Configure advanced Samba file sharing with domain integration, advanced security, user management, and performance optimization for enterprise file services

PREREQUISITES:
- RHEL 8/9 system with root access
- Understanding of SMB/CIFS protocols
- Basic knowledge of Windows networking
- Active Directory environment (optional)

LAB SCENARIO:
Deploy enterprise Samba file server with advanced features including domain integration, advanced ACLs, print sharing, performance tuning, and comprehensive security.

EQUIPMENT NEEDED:
- RHEL system (server: 192.168.1.10)
- Windows clients for testing
- Active Directory server (optional: 192.168.1.5)
- Storage for shared directories

PART F: CONFIGURE USER MANAGEMENT
----------------------------------

1. Sanity check
   pdbedit -L
getent passwd finuser hruser ituser

2. Create user management script:
   # vim /usr/local/bin/samba-user-mgmt.sh
   
   #!/bin/bash

set -e

ACTION="$1"
USER="$2"

case "$ACTION" in
  add)
    [ -z "$USER" ] && { echo "Username required"; exit 1; }
    id "$USER" &>/dev/null || useradd -M -s /sbin/nologin "$USER"
    smbpasswd -a "$USER"
    smbpasswd -e "$USER"
    echo "User $USER added and enabled"
    ;;
  delete)
    [ -z "$USER" ] && { echo "Username required"; exit 1; }
    smbpasswd -x "$USER"
    userdel "$USER" || true
    echo "User $USER deleted"
    ;;
  disable)
    [ -z "$USER" ] && { echo "Username required"; exit 1; }
    smbpasswd -d "$USER"
    echo "User $USER disabled"
    ;;
  enable)
    [ -z "$USER" ] && { echo "Username required"; exit 1; }
    smbpasswd -e "$USER"
    echo "User $USER enabled"
    ;;
  list)
    pdbedit -L
    ;;
  *)
    echo "Usage: $0 {add|delete|disable|enable|list} username"
    exit 1
    ;;
esac

chmod 700 /usr/local/bin/samba-user-mgmt.sh

/usr/local/bin/samba-user-mgmt.sh list

/usr/local/bin/samba-user-mgmt.sh disable ituser
/usr/local/bin/samba-user-mgmt.sh enable ituser

pdbedit -L
systemctl restart smb
systemctl status smb --no-pager

   #!/bin/bash

LOGFILE="/var/log/samba-monitor.log"

3. Create performance monitoring:
   # vim /usr/local/bin/samba-performance.sh
   
   #!/bin/bash

LOGFILE="/var/log/samba-performance.log"

echo "===== Samba Performance: $(date) =====" >> "$LOGFILE"

# Active connections
CONN=$(smbstatus -b 2>/dev/null | awk 'NR>4 {print}' | wc -l)
echo "Active connections: $CONN" >> "$LOGFILE"

# Active shares
SHARES=$(smbstatus -S 2>/dev/null | awk 'NR>4 {print}' | wc -l)
echo "Active shares: $SHARES" >> "$LOGFILE"

# smbd memory usage (KB)
MEM=$(ps -o rss= -C smbd | awk '{sum+=$1} END {print sum}')
echo "SMBD memory usage (KB): $MEM" >> "$LOGFILE"

echo "---" >> "$LOGFILE"

chmod 700 /usr/local/bin/samba-performance.sh

/usr/local/bin/samba-performance.sh

cat /var/log/samba-performance.log

PART I: CONFIGURE BACKUP AND RECOVERY
--------------------------------------

1. Create Samba configuration backup:
   # vim /usr/local/bin/backup-samba-config.sh
   
   #!/bin/bash
set -e

BACKUP_DIR="/opt/samba-backups"
DATE="$(date +%Y%m%d_%H%M%S)"

mkdir -p "$BACKUP_DIR"

# smb.conf
cp /etc/samba/smb.conf "$BACKUP_DIR/smb.conf.$DATE"

# Samba users DB
pdbedit -e tdbsam:/tmp/samba-users.tdb
cp /tmp/samba-users.tdb "$BACKUP_DIR/users.tdb.$DATE"
rm -f /tmp/samba-users.tdb

# Group mappings
net groupmap list > "$BACKUP_DIR/groupmap.$DATE"

echo "Config backup completed: $BACKUP_DIR"

chmod 700 /usr/local/bin/backup-samba-config.sh

/usr/local/bin/backup-samba-config.sh
ls -lh /opt/samba-backups

2. Create share data backup:
   # vim /usr/local/bin/backup-samba-shares.sh
   
   #!/bin/bash
set -e

BACKUP_DIR="/opt/share-backups"
DATE="$(date +%Y%m%d)"

mkdir -p "$BACKUP_DIR/$DATE"

rsync -a /srv/samba/finance/ "$BACKUP_DIR/$DATE/finance/"
rsync -a /srv/samba/hr/ "$BACKUP_DIR/$DATE/hr/"
rsync -a /srv/samba/it/ "$BACKUP_DIR/$DATE/it/"

tar czf "$BACKUP_DIR/shares-$DATE.tar.gz" -C "$BACKUP_DIR" "$DATE"
rm -rf "$BACKUP_DIR/$DATE"

echo "Share backup created: $BACKUP_DIR/shares-$DATE.tar.gz"

chmod 700 /usr/local/bin/backup-samba-shares.sh

/usr/local/bin/backup-samba-shares.sh
ls -lh /opt/share-backups

cp /opt/samba-backups/smb.conf.TIMESTAMP /etc/samba/smb.conf
systemctl restart smb

pdbedit -i tdbsam:/opt/samba-backups/users.tdb.TIMESTAMP

tar xzf /opt/share-backups/shares-YYYYMMDD.tar.gz -C /

2. Create timer for monitoring:
   # vim /etc/systemd/system/samba-monitor.timer
   
   [Unit]
Description=Run Samba Monitor every 5 minutes

[Timer]
OnCalendar=*:0/5
Persistent=true

[Install]
WantedBy=timers.target

3. Enable monitoring:
