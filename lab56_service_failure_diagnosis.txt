2. Create service analyzer tool:
   # vim /opt/service-diagnostics/scripts/service-analyzer.sh
   
   #!/bin/bash
#
# Service Failure Analysis Tool
#

DIAG_ROOT="/opt/service-diagnostics"
LOG_FILE="$DIAG_ROOT/logs/service-analysis.log"

mkdir -p "$DIAG_ROOT/logs"

log_message() {
    echo "$(date '+%Y-%m-%d %H:%M:%S') [$1] $2" | tee -a "$LOG_FILE"
}

analyze_service_status() {
    local service_name="$1"

    if [ -z "$service_name" ]; then
        log_message "ERROR" "Service name required"
        return 1
    fi

    log_message "INFO" "Analyzing service status: $service_name"

    systemctl status "$service_name" --no-pager > "$DIAG_ROOT/logs/status-${service_name}.txt" 2>&1
    systemctl show "$service_name" > "$DIAG_ROOT/logs/properties-${service_name}.txt"
    systemctl cat "$service_name" > "$DIAG_ROOT/logs/config-${service_name}.txt" 2>/dev/null
    journalctl -u "$service_name" --no-pager -n 100 > "$DIAG_ROOT/logs/journal-${service_name}.txt"

    if systemctl is-failed "$service_name" >/dev/null 2>&1; then
        systemctl status "$service_name" -l --no-pager > "$DIAG_ROOT/logs/failed-${service_name}.txt"
        log_message "WARN" "Service $service_name is in failed state"
    fi

    log_message "INFO" "Service status analysis completed for: $service_name"
}

analyze_all_failed_services() {
    log_message "INFO" "Analyzing all failed services"

    systemctl --failed --no-legend --no-pager | awk '{print $1}' > "$DIAG_ROOT/logs/failed-services-list.txt"

    while read -r service; do
        [ -n "$service" ] && analyze_service_status "$service"
    done < "$DIAG_ROOT/logs/failed-services-list.txt"

    systemctl --failed --no-pager > "$DIAG_ROOT/logs/failed-services-summary.txt"

    log_message "INFO" "Failed services analysis completed"
}

check_service_dependencies() {
    local service_name="$1"

    log_message "INFO" "Checking dependencies for service: $service_name"

    systemctl list-dependencies "$service_name" --no-pager > "$DIAG_ROOT/logs/deps-${service_name}.txt"
    systemctl list-dependencies "$service_name" --reverse --no-pager > "$DIAG_ROOT/logs/rdeps-${service_name}.txt"
    systemctl show "$service_name" -p Requires,Wants,After,Before > "$DIAG_ROOT/logs/dep-details-${service_name}.txt"

    local deps=$(systemctl show "$service_name" -p Requires --value)
    for dep in $deps; do
        if systemctl is-failed "$dep" >/dev/null 2>&1; then
            log_message "WARN" "Dependency $dep is failed for service $service_name"
        fi
    done

    log_message "INFO" "Dependency check completed for: $service_name"
}

# ---- argument handler ----
case "$1" in
    analyze_service_status)
        analyze_service_status "$2"
        ;;
    analyze_all_failed_services)
        analyze_all_failed_services
        ;;
    check_service_dependencies)
        check_service_dependencies "$2"
        ;;
    *)
        echo "Usage:"
        echo "$0 analyze_service_status <service>"
        echo "$0 analyze_all_failed_services"
        echo "$0 check_service_dependencies <service>"
        exit 1
        ;;
esac

3. Create service health monitor:
   # vim /opt/service-diagnostics/scripts/service-health-monitor.sh
   
   #!/bin/bash
#
# Service Health Monitoring Tool
#

DIAG_ROOT="/opt/service-diagnostics"
LOG_FILE="$DIAG_ROOT/logs/service-health.log"

ANALYZER="/opt/service-diagnostics/scripts/service-analyzer.sh"
RECOVERY="/opt/service-diagnostics/scripts/service-recovery.sh"

mkdir -p "$DIAG_ROOT/logs"

log_message() {
    echo "$(date '+%Y-%m-%d %H:%M:%S') [$1] $2" | tee -a "$LOG_FILE"
}

monitor_service_health() {
    local service_name="$1"
    local check_interval="${2:-60}"

    if [ -z "$service_name" ]; then
        echo "Usage: $0 monitor <service> [interval]"
        exit 1
    fi

    log_message "INFO" "Starting health monitoring for service: $service_name"

    while true; do
        timestamp=$(date '+%Y-%m-%d %H:%M:%S')
        status=$(systemctl is-active "$service_name" 2>/dev/null)
        enabled=$(systemctl is-enabled "$service_name" 2>/dev/null)

        echo "$timestamp,$service_name,$status,$enabled" >> "$DIAG_ROOT/logs/health-monitor.csv"

        if [ "$status" != "active" ]; then
            log_message "WARN" "Service $service_name is not active: $status"

            if [ -x "$ANALYZER" ]; then
                "$ANALYZER" analyze_service_status "$service_name"
            fi

            if [ -x "$RECOVERY" ]; then
                "$RECOVERY" attempt_service_recovery "$service_name" basic
            fi
        fi

        sleep "$check_interval"
    done
}

check_service_resources() {
    local service_name="$1"

    log_message "INFO" "Checking resource usage for service: $service_name"

    main_pid=$(systemctl show "$service_name" -p MainPID --value)

    if [ "$main_pid" != "0" ] && [ -n "$main_pid" ]; then
        ps -p "$main_pid" -o pid,ppid,pcpu,pmem,vsz,rss,comm > "$DIAG_ROOT/logs/resources-${service_name}.txt"
        pstree -p "$main_pid" >> "$DIAG_ROOT/logs/resources-${service_name}.txt" 2>/dev/null
        ls -la "/proc/$main_pid/fd/" > "$DIAG_ROOT/logs/fds-${service_name}.txt" 2>/dev/null
        cat "/proc/$main_pid/maps" > "$DIAG_ROOT/logs/maps-${service_name}.txt" 2>/dev/null
    else
        log_message "WARN" "No main PID found for service: $service_name"
    fi

    systemctl show "$service_name" -p MemoryLimit,CPUQuota,TasksMax > "$DIAG_ROOT/logs/limits-${service_name}.txt"

    log_message "INFO" "Resource check completed for: $service_name"
}

case "$1" in
    monitor)
        monitor_service_health "$2" "$3"
        ;;
    resources)
        check_service_resources "$2"
        ;;
    *)
        echo "Usage:"
        echo "$0 monitor <service> [interval]"
        echo "$0 resources <service>"
        exit 1
        ;;
esac

1. Create systemd diagnostics tool:

   cat << 'EOF' > /opt/service-diagnostics/scripts/systemd-diagnostics.sh
#!/bin/bash

DIAG_ROOT="/opt/service-diagnostics"
LOG_FILE="$DIAG_ROOT/logs/systemd-diagnostics.log"

mkdir -p "$DIAG_ROOT/logs"

log_message() {
    echo "$(date '+%Y-%m-%d %H:%M:%S') [$1] $2" | tee -a "$LOG_FILE"
}

analyze_systemd_config() {
    service_name="$1"
    [ -z "$service_name" ] && { echo "service required"; exit 1; }

    log_message "INFO" "Analyzing systemd configuration for: $service_name"

    unit_file=$(systemctl show "$service_name" -p FragmentPath --value)

    if [ -n "$unit_file" ] && [ -f "$unit_file" ]; then
        cat "$unit_file" > "$DIAG_ROOT/logs/unit-file-${service_name}.txt"
        systemd-analyze verify "$unit_file" > "$DIAG_ROOT/logs/verify-${service_name}.txt" 2>&1
    fi

    systemctl show "$service_name" -p DropInPaths --value > "$DIAG_ROOT/logs/dropins-${service_name}.txt"
    systemctl show-environment > "$DIAG_ROOT/logs/systemd-environment.txt"
    systemctl list-units --all | grep "$service_name" > "$DIAG_ROOT/logs/conflicts-${service_name}.txt"

    log_message "INFO" "Systemd configuration analysis completed"
}

check_startup_sequence() {
    service_name="$1"
    [ -z "$service_name" ] && exit 1

    log_message "INFO" "Checking startup sequence for: $service_name"

    systemd-analyze critical-chain "$service_name" > "$DIAG_ROOT/logs/critical-chain-${service_name}.txt" 2>/dev/null
    systemd-analyze blame | grep "$service_name" > "$DIAG_ROOT/logs/timing-${service_name}.txt"
    systemctl show "$service_name" -p After,Before,Wants,Requires > "$DIAG_ROOT/logs/ordering-${service_name}.txt"

    log_message "INFO" "Startup sequence check completed"
}

fix_systemd_issues() {
    service_name="$1"
    [ -z "$service_name" ] && exit 1

    log_message "INFO" "Attempting to fix systemd issues for: $service_name"

    systemctl daemon-reload
    systemctl reset-failed "$service_name"

    if systemctl is-masked "$service_name" >/dev/null 2>&1; then
        log_message "WARN" "Service masked — unmasking"
        systemctl unmask "$service_name"
    fi

    systemctl enable "$service_name" >/dev/null 2>&1
    systemctl start "$service_name"
}

case "$1" in
    analyze) analyze_systemd_config "$2" ;;
    startup) check_startup_sequence "$2" ;;
    fix) fix_systemd_issues "$2" ;;
    *) echo "Usage: $0 {analyze|startup|fix} servicename"; exit 1 ;;
esac
EOF

2. Create service configuration validator:
   cat > /opt/service-diagnostics/scripts/config-validator.sh << 'EOF'
#!/bin/bash
#
# Service Configuration Validator
#

DIAG_ROOT="/opt/service-diagnostics"
LOG_FILE="$DIAG_ROOT/logs/config-validator.log"

log_message() {
    echo "$(date '+%Y-%m-%d %H:%M:%S') [$1] $2" | tee -a "$LOG_FILE"
}

validate_service_config() {
    local service_name="$1"

    log_message "INFO" "Validating configuration for service: $service_name"

    local validation_errors=0

    # ---- UNIT FILE SYNTAX CHECK (RHEL9 SAFE) ----
    local unit_file=$(systemctl show "$service_name" -p FragmentPath --value)
    if [ -n "$unit_file" ] && [ -f "$unit_file" ]; then
        verify_output=$(systemd-analyze verify "$unit_file" 2>&1)

        if echo "$verify_output" | grep -Ei "Failed to parse|Invalid argument|syntax error" >/dev/null; then
            log_message "ERROR" "Unit file syntax error: $unit_file"
            validation_errors=$((validation_errors + 1))
        else
            log_message "INFO" "Unit file syntax OK: $unit_file"
        fi
    fi

    # ---- EXECUTABLE CHECK (RHEL9 FORMAT) ----
    local exec_start=$(systemctl show "$service_name" -p ExecStart --value)
    if [ -n "$exec_start" ]; then
        local executable=$(echo "$exec_start" | grep -oP 'path=\K[^ ;]+')
        if [ -n "$executable" ] && [ ! -x "$executable" ]; then
            log_message "ERROR" "Executable not found or not executable: $executable"
            validation_errors=$((validation_errors + 1))
        fi
    fi

    # ---- USER CHECK ----
    local user=$(systemctl show "$service_name" -p User --value)
    if [ -n "$user" ] && ! id "$user" >/dev/null 2>&1; then
        log_message "ERROR" "Service user does not exist: $user"
        validation_errors=$((validation_errors + 1))
    fi

    # ---- GROUP CHECK ----
    local group=$(systemctl show "$service_name" -p Group --value)
    if [ -n "$group" ] && ! getent group "$group" >/dev/null 2>&1; then
        log_message "ERROR" "Service group does not exist: $group"
        validation_errors=$((validation_errors + 1))
    fi

    # ---- WORKDIR CHECK ----
    local working_dir=$(systemctl show "$service_name" -p WorkingDirectory --value)
    if [ -n "$working_dir" ] && [ ! -d "$working_dir" ]; then
        log_message "ERROR" "Working directory does not exist: $working_dir"
        validation_errors=$((validation_errors + 1))
    fi

    return $validation_errors
}

case "$1" in
    validate)
        validate_service_config "$2"
        echo "Exit Code: $?"
        ;;
    *)
        echo "Usage: $0 validate <service>"
        exit 1
        ;;
esac
EOF

1. Create application diagnostics tool:
   cat << 'EOF' > /opt/service-diagnostics/scripts/app-diagnostics.sh
#!/bin/bash

DIAG_ROOT="/opt/service-diagnostics"
LOG_FILE="$DIAG_ROOT/logs/app-diagnostics.log"

mkdir -p "$DIAG_ROOT/logs"

log_message() {
    echo "$(date '+%Y-%m-%d %H:%M:%S') [$1] $2" | tee -a "$LOG_FILE"
}

command_exists() {
    command -v "$1" >/dev/null 2>&1
}

diagnose_web_service() {
    service_name="$1"
    log_message "INFO" "Diagnosing web service: $service_name"

    ss -tlnp | grep -E "(httpd|nginx|apache)" > "$DIAG_ROOT/logs/web-ports-${service_name}.txt" 2>/dev/null

    ports=$(ss -tlnp | grep -E "(httpd|nginx|apache)" | awk '{print $4}' | awk -F: '{print $NF}')

    for port in $ports; do
        if command_exists curl; then
            curl -I "http://localhost:$port" > "$DIAG_ROOT/logs/http-test-${port}.txt" 2>&1
        fi
    done

    if command_exists httpd && [ "$service_name" = "httpd" ]; then
        httpd -t > "$DIAG_ROOT/logs/httpd-config-test.txt" 2>&1
    fi

    if command_exists nginx && [ "$service_name" = "nginx" ]; then
        nginx -t > "$DIAG_ROOT/logs/nginx-config-test.txt" 2>&1
    fi

    log_message "INFO" "Web service diagnosis completed"
}

diagnose_database_service() {
    service_name="$1"
    log_message "INFO" "Diagnosing database service: $service_name"

    ss -tlnp | grep -E "(mysql|mariadb|postgres)" > "$DIAG_ROOT/logs/db-ports-${service_name}.txt" 2>/dev/null
    ps aux | grep -E "(mysql|mariadb|postgres)" > "$DIAG_ROOT/logs/db-processes-${service_name}.txt"

    if command_exists mysqladmin; then
        mysqladmin ping > "$DIAG_ROOT/logs/mysql-ping.txt" 2>&1
    fi

    if command_exists pg_isready; then
        pg_isready > "$DIAG_ROOT/logs/postgres-ready.txt" 2>&1
    fi

    log_message "INFO" "Database service diagnosis completed"
}

diagnose_mail_service() {
    service_name="$1"
    log_message "INFO" "Diagnosing mail service: $service_name"

    ss -tlnp | grep -E ":25|:587|:993|:995" > "$DIAG_ROOT/logs/mail-ports-${service_name}.txt" 2>/dev/null

    if command_exists postqueue; then
        postqueue -p > "$DIAG_ROOT/logs/postfix-queue.txt" 2>&1
        postconf -n > "$DIAG_ROOT/logs/postfix-config.txt" 2>&1
    fi

    if command_exists mailq; then
        mailq > "$DIAG_ROOT/logs/sendmail-queue.txt" 2>&1
    fi

    log_message "INFO" "Mail service diagnosis completed"
}

case "$1" in
    web) diagnose_web_service "$2" ;;
    db) diagnose_database_service "$2" ;;
    mail) diagnose_mail_service "$2" ;;
    *) echo "Usage: $0 {web|db|mail} servicename"; exit 1 ;;
esac
EOF

2. Create service recovery procedures:
   cat << 'EOF' > /opt/service-diagnostics/scripts/service-recovery.sh
#!/bin/bash

DIAG_ROOT="/opt/service-diagnostics"
LOG_FILE="$DIAG_ROOT/logs/service-recovery.log"

mkdir -p "$DIAG_ROOT/logs"

log_message() {
    echo "$(date '+%Y-%m-%d %H:%M:%S') [$1] $2" | tee -a "$LOG_FILE"
}

service_exists() {
    systemctl list-unit-files | grep -q "^$1"
}

basic_recovery() {
    service_name="$1"

    log_message "INFO" "Basic recovery: $service_name"

    systemctl stop "$service_name" 2>/dev/null
    systemctl reset-failed "$service_name" 2>/dev/null
    systemctl daemon-reload

    if systemctl start "$service_name"; then
        log_message "INFO" "Service started successfully"
        return 0
    else
        log_message "WARN" "Basic recovery failed"
        return 1
    fi
}

advanced_recovery() {
    service_name="$1"

    log_message "INFO" "Advanced recovery: $service_name"

    # remove stale pid files
    pidfile=$(systemctl show "$service_name" -p PIDFile --value)
    if [ -n "$pidfile" ] && [ -f "$pidfile" ]; then
        rm -f "$pidfile"
        log_message "INFO" "Removed stale PID file"
    fi

    # clean runtime directory
    runtime_dir="/run/$service_name"
    [ -d "$runtime_dir" ] && rm -rf "$runtime_dir"

    basic_recovery "$service_name"
}

full_recovery() {
    service_name="$1"

    log_message "INFO" "Full recovery: $service_name"

    # unmask if masked
    if systemctl is-masked "$service_name" >/dev/null 2>&1; then
        systemctl unmask "$service_name"
        log_message "INFO" "Service unmasked"
    fi

    # re-enable
    systemctl enable "$service_name" >/dev/null 2>&1

    advanced_recovery "$service_name"
}

attempt_service_recovery() {
    service_name="$1"
    level="${2:-basic}"

    if ! service_exists "$service_name"; then
        log_message "ERROR" "Service does not exist: $service_name"
        exit 1
    fi

    case "$level" in
        basic) basic_recovery "$service_name" ;;
        advanced) advanced_recovery "$service_name" ;;
        full) full_recovery "$service_name" ;;
        *) log_message "ERROR" "Invalid recovery level"; exit 1 ;;
    esac
}

case "$1" in
    attempt_service_recovery)
        attempt_service_recovery "$2" "$3"
        ;;
    *)
        echo "Usage: $0 attempt_service_recovery <service> [basic|advanced|full]"
        exit 1
        ;;
esac
EOF

PART D: DEPENDENCY AND CONFLICT RESOLUTION
-------------------------------------------

1. Create dependency analyzer:
   cat << 'EOF' > /opt/service-diagnostics/scripts/dependency-analyzer.sh
#!/bin/bash

DIAG_ROOT="/opt/service-diagnostics"
LOG_FILE="$DIAG_ROOT/logs/dependency-analyzer.log"

mkdir -p "$DIAG_ROOT/logs"

log_message() {
    echo "$(date '+%Y-%m-%d %H:%M:%S') [$1] $2" | tee -a "$LOG_FILE"
}

analyze_dependency_chain() {
    service_name="$1"
    [ -z "$service_name" ] && { echo "service required"; exit 1; }

    log_message "INFO" "Analyzing dependency chain for: $service_name"

    systemctl list-dependencies "$service_name" --all --plain > "$DIAG_ROOT/logs/dep-graph-${service_name}.txt"

    deps=$(systemctl show "$service_name" -p Requires,Wants --value | tr ' ' '\n')

    for dep in $deps; do
        [ -z "$dep" ] && continue

        status=$(systemctl is-active "$dep" 2>/dev/null)
        echo "$dep : $status" >> "$DIAG_ROOT/logs/dep-status-${service_name}.txt"

        if [ "$status" = "failed" ]; then
            log_message "WARN" "Dependency failed: $dep"
        fi
    done

    log_message "INFO" "Dependency chain analysis completed"
}

resolve_dependency_conflicts() {
    service_name="$1"

    log_message "INFO" "Resolving dependencies: $service_name"

    deps=$(systemctl show "$service_name" -p Requires --value)

    for dep in $deps; do
        if ! systemctl is-active "$dep" >/dev/null 2>&1; then
            log_message "INFO" "Starting dependency: $dep"
            systemctl start "$dep"
        fi
    done

    log_message "INFO" "Dependency resolution completed"
}

case "$1" in
    analyze) analyze_dependency_chain "$2" ;;
    fix) resolve_dependency_conflicts "$2" ;;
    *) echo "Usage: $0 {analyze|fix} servicename"; exit 1 ;;
esac
EOF

1. Create resource conflict analyzer:
   cat << 'EOF' > /opt/service-diagnostics/scripts/resource-analyzer.sh
#!/bin/bash

DIAG_ROOT="/opt/service-diagnostics"
LOG_FILE="$DIAG_ROOT/logs/resource-analyzer.log"

mkdir -p "$DIAG_ROOT/logs"

log_message() {
    echo "$(date '+%Y-%m-%d %H:%M:%S') [$1] $2" | tee -a "$LOG_FILE"
}

check_port_conflicts() {
    service_name="$1"
    log_message "INFO" "Checking port conflicts for: $service_name"

    ports=$(ss -tulpn | grep "$service_name" | awk '{print $5}' | awk -F: '{print $NF}')

    for port in $ports; do
        owner=$(ss -tulpn | grep ":$port " | awk '{print $7}')
        echo "Port $port -> $owner" >> "$DIAG_ROOT/logs/port-conflicts-${service_name}.txt"
    done

    log_message "INFO" "Port conflict check completed"
}

check_file_conflicts() {
    service_name="$1"
    log_message "INFO" "Checking file conflicts for: $service_name"

    pidfile=$(systemctl show "$service_name" -p PIDFile --value)

    if [ -n "$pidfile" ] && [ -f "$pidfile" ]; then
        pid=$(cat "$pidfile")
        if ! ps -p "$pid" >/dev/null 2>&1; then
            echo "STALE PID FILE: $pidfile" >> "$DIAG_ROOT/logs/file-conflicts-${service_name}.txt"
        fi
    fi

    find /run /var/lock -name "*${service_name}*" 2>/dev/null >> "$DIAG_ROOT/logs/file-conflicts-${service_name}.txt"

    log_message "INFO" "File conflict check completed"
}

case "$1" in
    ports) check_port_conflicts "$2" ;;
    files) check_file_conflicts "$2" ;;
    *) echo "Usage: $0 {ports|files} servicename"; exit 1 ;;
esac
EOF

echo '        continue' >> /opt/service-diagnostics/scripts/auto-recovery.sh
echo '    fi' >> /opt/service-diagnostics/scripts/auto-recovery.sh
echo '' >> /opt/service-diagnostics/scripts/auto-recovery.sh
echo '    while read -r service level attempts; do' >> /opt/service-diagnostics/scripts/auto-recovery.sh
echo '        [[ "$service" =~ ^#.*$ || -z "$service" ]] && continue' >> /opt/service-diagnostics/scripts/auto-recovery.sh
echo '' >> /opt/service-diagnostics/scripts/auto-recovery.sh
echo '        if ! systemctl is-active "$service" >/dev/null 2>&1; then' >> /opt/service-diagnostics/scripts/auto-recovery.sh
echo '            log "Service down: $service (level=$level)"' >> /opt/service-diagnostics/scripts/auto-recovery.sh
echo '' >> /opt/service-diagnostics/scripts/auto-recovery.sh
echo '            count=0' >> /opt/service-diagnostics/scripts/auto-recovery.sh
echo '            max=${attempts:-3}' >> /opt/service-diagnostics/scripts/auto-recovery.sh
echo '' >> /opt/service-diagnostics/scripts/auto-recovery.sh
echo '            while [ $count -lt $max ]; do' >> /opt/service-diagnostics/scripts/auto-recovery.sh
echo '                if "$RECOVERY" attempt_service_recovery "$service" "$level"; then' >> /opt/service-diagnostics/scripts/auto-recovery.sh
echo '                    log "Recovered: $service"' >> /opt/service-diagnostics/scripts/auto-recovery.sh
echo '                    break' >> /opt/service-diagnostics/scripts/auto-recovery.sh
echo '                fi' >> /opt/service-diagnostics/scripts/auto-recovery.sh
echo '                count=$((count+1))' >> /opt/service-diagnostics/scripts/auto-recovery.sh
echo '                sleep 15' >> /opt/service-diagnostics/scripts/auto-recovery.sh
echo '            done' >> /opt/service-diagnostics/scripts/auto-recovery.sh
echo '        fi' >> /opt/service-diagnostics/scripts/auto-recovery.sh
echo '    done < "$CONFIG"' >> /opt/service-diagnostics/scripts/auto-recovery.sh
echo '' >> /opt/service-diagnostics/scripts/auto-recovery.sh
echo '    sleep 60' >> /opt/service-diagnostics/scripts/auto-recovery.sh
echo 'done' >> /opt/service-diagnostics/scripts/auto-recovery.sh

chmod +x /opt/service-diagnostics/scripts/auto-recovery.sh

2. Create recovery configuration:
   cat << 'EOF' > /opt/service-diagnostics/configs/auto-recovery.conf
# service   level     attempts
sshd        advanced  2
systemd-journald basic 2
EOF

3. systemd monitoring service

cat << 'EOF' > /etc/systemd/system/service-monitor.service
[Unit]
Description=Service Failure Monitor
After=multi-user.target

[Service]
Type=simple
ExecStart=/opt/service-diagnostics/scripts/auto-recovery.sh
Restart=always
RestartSec=20

# Safety: runaway restart protection
StartLimitIntervalSec=0

[Install]
WantedBy=multi-user.target
EOF

systemctl daemon-reload
systemctl enable --now service-monitor.service
