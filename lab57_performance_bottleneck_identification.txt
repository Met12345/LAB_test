2. Common logging + persistence library

cat > /opt/performance-diagnostics/scripts/common.sh << 'EOF'
#!/bin/bash

PERF_ROOT="/opt/performance-diagnostics"
LOG_FILE="$PERF_ROOT/logs/performance-analysis.log"
STATE_DIR="$PERF_ROOT/state"

mkdir -p "$PERF_ROOT/logs" "$STATE_DIR"

log_message() {
    local level="$1"
    local message="$2"
    echo "$(date '+%Y-%m-%d %H:%M:%S') [$level] $message" | tee -a "$LOG_FILE"
}

# safe numeric compare (bc empty output protection)
safe_compare() {
    local expr="$1"
    local result
    result=$(echo "$expr" | bc 2>/dev/null)
    [[ "$result" == "1" ]] && echo 1 || echo 0
}

# persistence detection (trend based alert)
check_persistent_issue() {
    local name="$1"
    local expr="$2"
    local threshold="${3:-5}"

    local file="$STATE_DIR/${name}.cnt"
    [ -f "$file" ] || echo 0 > "$file"

    local status=$(safe_compare "$expr")

    if [[ "$status" -eq 1 ]]; then
        cnt=$(cat "$file")
        cnt=$((cnt+1))
        echo $cnt > "$file"
        if [[ "$cnt" -ge "$threshold" ]]; then
            log_message "WARN" "Persistent issue detected: $name"
        fi
    else
        echo 0 > "$file"
    fi
}
EOF

3. Create performance analyzer tool:
   cat > /opt/performance-diagnostics/scripts/performance-analyzer.sh << 'EOF'
#!/bin/bash
source /opt/performance-diagnostics/scripts/common.sh

system_overview() {

    log_message INFO "Gathering system overview"

    uname -a > "$PERF_ROOT/logs/system-info.txt"
    cat /proc/cpuinfo > "$PERF_ROOT/logs/cpu-info.txt"
    cat /proc/meminfo > "$PERF_ROOT/logs/memory-info.txt"

    uptime > "$PERF_ROOT/logs/uptime.txt"
    cat /proc/loadavg > "$PERF_ROOT/logs/loadavg.txt"

    ps aux --sort=-%cpu | head -20 > "$PERF_ROOT/logs/top-cpu-processes.txt"
    ps aux --sort=-%mem | head -20 > "$PERF_ROOT/logs/top-memory-processes.txt"

    vmstat 1 3 > "$PERF_ROOT/logs/vmstat-snapshot.txt"
    iostat -x 1 3 > "$PERF_ROOT/logs/iostat-snapshot.txt"

    log_message INFO "System overview completed"
}

identify_bottlenecks() {

    log_message INFO "Checking basic system bottlenecks"

    load=$(awk '{print $1}' /proc/loadavg)
    cores=$(nproc)

    check_persistent_issue "cpu_load_high" "$load > $cores"

    mem_used=$(free | awk '/Mem/ {printf "%.0f",$3/$2*100}')
    check_persistent_issue "memory_pressure" "$mem_used > 90"

    io_wait=$(vmstat 1 2 | tail -1 | awk '{print $16}')
    check_persistent_issue "io_wait_high" "$io_wait > 20"

    log_message INFO "Basic bottleneck scan completed"
}

case "$1" in
    system_overview) system_overview ;;
    identify_bottlenecks) identify_bottlenecks ;;
    *) echo "Usage: $0 {system_overview|identify_bottlenecks}" ;;
esac
EOF

#!/bin/bash
source /opt/performance-diagnostics/scripts/common.sh

collect_baseline() {

    duration="${1:-60}"
    name="${2:-baseline_$(date +%F_%H%M)}"
    dir="$PERF_ROOT/baselines/$name"

    mkdir -p "$dir"

    log_message INFO "Collecting baseline $name"

    sar -u 1 $duration > "$dir/cpu.txt" &
    sar -r 1 $duration > "$dir/memory.txt" &
    sar -d 1 $duration > "$dir/disk.txt" &
    sar -n DEV 1 $duration > "$dir/network.txt" &

    wait
    log_message INFO "Baseline collection finished"
}

case "$1" in
    collect_baseline) collect_baseline "$2" "$3" ;;
    *) echo "Usage: $0 collect_baseline [duration] [name]" ;;
esac
EOF

1. Create CPU analyzer:
   cat > /opt/performance-diagnostics/scripts/cpu-analyzer.sh << 'EOF'
#!/bin/bash
source /opt/performance-diagnostics/scripts/common.sh

CPU_DIR="$PERF_ROOT/analysis/cpu"
mkdir -p "$CPU_DIR"

#######################################
# CPU PERFORMANCE ANALYSIS
#######################################
analyze_cpu_performance() {

    log_message INFO "Analyzing CPU performance"

    sar -u 1 5 > "$CPU_DIR/cpu-utilization.txt"
    mpstat -P ALL 1 5 > "$CPU_DIR/per-core.txt"
    sar -w 1 5 > "$CPU_DIR/context-switch.txt"
    ps -eo pid,comm,%cpu --sort=-%cpu | head -20 > "$CPU_DIR/top-cpu.txt"

    log_message INFO "CPU performance analysis completed"
}

#######################################
# CPU BOTTLENECK DETECTION
#######################################
identify_cpu_bottlenecks() {

    log_message INFO "Checking CPU bottlenecks"

    load=$(awk '{print $1}' /proc/loadavg)
    cores=$(nproc)

    # persistent high load
    check_persistent_issue "cpu_overload" "$load > $cores"

    # steal time (virtual machines)
    steal=$(sar -u 1 1 | awk '/Average/ {print $6}')
    steal=${steal:-0}
    check_persistent_issue "cpu_steal_time" "$steal > 10"

    # top process hog
    hog=$(ps -eo %cpu --sort=-%cpu | awk 'NR==2{print int($1)}')
    hog=${hog:-0}
    check_persistent_issue "single_process_cpu_hog" "$hog > 80"

    log_message INFO "CPU bottleneck scan completed"
}

#######################################
# CPU STRESS TEST
#######################################
cpu_stress_test() {

    duration="${1:-60}"
    cores=$(nproc)

    log_message INFO "Starting CPU stress test (${cores} cores for ${duration}s)"

    stress-ng --cpu $cores --timeout ${duration}s --metrics-brief > "$CPU_DIR/stress-test.txt" 2>&1

    log_message INFO "CPU stress test finished"
}

case "$1" in
    analyze_cpu_performance) analyze_cpu_performance ;;
    identify_cpu_bottlenecks) identify_cpu_bottlenecks ;;
    cpu_stress_test) cpu_stress_test "$2" ;;
    *) echo "Usage: $0 {analyze_cpu_performance|identify_cpu_bottlenecks|cpu_stress_test}" ;;
esac
EOF

chmod +x /opt/performance-diagnostics/scripts/cpu-analyzer.sh

/opt/performance-diagnostics/scripts/cpu-analyzer.sh identify_cpu_bottlenecks

#!/bin/bash
source /opt/performance-diagnostics/scripts/common.sh

MEM_DIR="$PERF_ROOT/analysis/memory"
mkdir -p "$MEM_DIR"

#######################################
# MEMORY PERFORMANCE ANALYSIS
#######################################
analyze_memory_performance() {

    log_message INFO "Analyzing memory performance"

    free -h > "$MEM_DIR/memory-usage.txt"
    cat /proc/meminfo > "$MEM_DIR/meminfo.txt"
    sar -r 1 5 > "$MEM_DIR/memory-stats.txt"
    sar -S 1 5 > "$MEM_DIR/swap-stats.txt"
    ps -eo pid,comm,%mem --sort=-%mem | head -20 > "$MEM_DIR/top-memory.txt"

    log_message INFO "Memory performance analysis completed"
}

#######################################
# MEMORY BOTTLENECK DETECTION
#######################################
identify_memory_bottlenecks() {

    log_message INFO "Checking memory bottlenecks"

    mem_used=$(free | awk '/Mem/ {printf "%.0f",$3/$2*100}')
    mem_used=${mem_used:-0}
    check_persistent_issue "memory_pressure" "$mem_used > 90"

    swap_used=$(free | awk '/Swap/ {print $3}')

    check_persistent_issue "swap_usage" "$swap_used > 0"

    oom=$(dmesg | grep -i "killed process" | wc -l)
    oom=${oom:-0}
    check_persistent_issue "oom_killer_activity" "$oom > 0"

    log_message INFO "Memory bottleneck scan completed"

    log_message INFO "Memory stress test finished"

1. Create disk I/O analyzer:
   cat > /opt/performance-diagnostics/scripts/disk-analyzer.sh << 'EOF'
#!/bin/bash
source /opt/performance-diagnostics/scripts/common.sh

DISK_DIR="$PERF_ROOT/analysis/disk"
mkdir -p "$DISK_DIR"

#######################################
# DISK PERFORMANCE ANALYSIS
#######################################
analyze_disk_performance() {

    log_message INFO "Analyzing disk performance"

    iostat -x 1 5 > "$DISK_DIR/iostat.txt"
    sar -d 1 5 > "$DISK_DIR/disk-stats.txt"
    df -h > "$DISK_DIR/disk-usage.txt"
    lsblk -f > "$DISK_DIR/block-devices.txt"

    log_message INFO "Disk performance analysis completed"
}

#######################################
# DISK BOTTLENECK DETECTION
#######################################
identify_disk_bottlenecks() {

    log_message INFO "Checking disk bottlenecks"

    io_wait=$(vmstat 1 2 | tail -1 | awk '{print $16}')
    io_wait=${io_wait:-0}
    check_persistent_issue "disk_io_wait" "$io_wait > 20"

    full_fs=$(df -h | awk 'NR>1 {gsub("%","",$5); if($5>90) print 1; else print 0}' | grep -c 1)
    full_fs=${full_fs:-0}
    check_persistent_issue "filesystem_full" "$full_fs > 0"

    log_message INFO "Disk bottleneck scan completed"
}

#######################################
# DISK STRESS TEST
#######################################
disk_stress_test() {

    duration="${1:-60}"
    testfile="/tmp/disk_stress_test"

    log_message INFO "Starting disk stress test (${duration}s)"

    dd if=/dev/zero of=$testfile bs=1M count=1024 oflag=direct status=none
    timeout ${duration}s dd if=$testfile of=/dev/null bs=1M iflag=direct status=none

    rm -f $testfile

    log_message INFO "Disk stress test finished"
}

case "$1" in
    analyze_disk_performance) analyze_disk_performance ;;
    identify_disk_bottlenecks) identify_disk_bottlenecks ;;
    disk_stress_test) disk_stress_test "$2" ;;
    *) echo "Usage: $0 {analyze_disk_performance|identify_disk_bottlenecks|disk_stress_test}" ;;
esac
EOF

chmod +x /opt/performance-diagnostics/scripts/disk-analyzer.sh

1. Create network analyzer:
   cat > /opt/performance-diagnostics/scripts/network-analyzer.sh << 'EOF'
#!/bin/bash
source /opt/performance-diagnostics/scripts/common.sh

NET_DIR="$PERF_ROOT/analysis/network"
mkdir -p "$NET_DIR"

get_iface() {
    ip route | awk '/default/ {print $5; exit}'
}

#######################################
# NETWORK PERFORMANCE ANALYSIS
#######################################
analyze_network_performance() {

    log_message INFO "Analyzing network performance"

    sar -n DEV 1 5 > "$NET_DIR/interface-stats.txt"
    sar -n EDEV 1 5 > "$NET_DIR/network-errors.txt"
    ss -s > "$NET_DIR/socket-summary.txt"
    ss -tuln > "$NET_DIR/listening-ports.txt"
    ip addr show > "$NET_DIR/interface-info.txt"

    log_message INFO "Network performance analysis completed"
}

#######################################
# NETWORK BOTTLENECK DETECTION
#######################################
identify_network_bottlenecks() {

    log_message INFO "Checking network bottlenecks"

    IFACE=$(get_iface)

    RX=$(sar -n DEV 1 1 | awk -v i="$IFACE" '$2==i {print int($5)}')
    TX=$(sar -n DEV 1 1 | awk -v i="$IFACE" '$2==i {print int($6)}')
    ERR=$(sar -n EDEV 1 1 | awk -v i="$IFACE" '$2==i {print $3+$4+$5+$6}')
    CONN=$(ss -tun | wc -l)

    RX=${RX:-0}
    TX=${TX:-0}
    ERR=${ERR:-0}
    CONN=${CONN:-0}

    # thresholds tuned for localhost heavy testing
    check_persistent_issue "high_rx_usage" "$RX > 50000"
    check_persistent_issue "high_tx_usage" "$TX > 50000"
    check_persistent_issue "network_errors" "$ERR > 0"
    check_persistent_issue "connection_flood" "$CONN > 800"

    log_message INFO "Network bottleneck scan completed"
}

case "$1" in
    analyze_network_performance) analyze_network_performance ;;
    identify_network_bottlenecks) identify_network_bottlenecks ;;
    *) echo "Usage: $0 {analyze_network_performance|identify_network_bottlenecks}" ;;
esac
EOF

chmod +x /opt/performance-diagnostics/scripts/network-analyzer.sh

/opt/performance-diagnostics/scripts/network-analyzer.sh analyze_network_performance
/opt/performance-diagnostics/scripts/network-analyzer.sh identify_network_bottlenecks
