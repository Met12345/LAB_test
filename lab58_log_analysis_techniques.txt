   cat > /opt/log-analysis/scripts/log-analyzer.sh << 'EOF'
#!/bin/bash
#
# Master Log Analysis Tool
#

LOG_ANALYSIS_ROOT="/opt/log-analysis"
ANALYSIS_LOG="$LOG_ANALYSIS_ROOT/logs/analysis.log"

log_message() {
    echo "$(date '+%Y-%m-%d %H:%M:%S') [$1] $2" | tee -a "$ANALYSIS_LOG"
}

# Gather system logs
gather_system_logs() {
    local days_back="${1:-7}"
    
    log_message "INFO" "Gathering system logs for last $days_back days"
    
    journalctl --since="$days_back days ago" --no-pager > "$LOG_ANALYSIS_ROOT/logs/journal-${days_back}days.log"

    find /var/log -name "*.log" -mtime -"$days_back" -exec cp {} "$LOG_ANALYSIS_ROOT/logs/" \; 2>/dev/null

    [ -f /var/log/secure ] && cp /var/log/secure "$LOG_ANALYSIS_ROOT/logs/secure.log"
    [ -f /var/log/audit/audit.log ] && cp /var/log/audit/audit.log "$LOG_ANALYSIS_ROOT/logs/audit.log"

    log_message "INFO" "Log gathering completed"
}

# Basic log statistics
generate_log_statistics() {
    log_message "INFO" "Generating log statistics"
    
    local stats_file="$LOG_ANALYSIS_ROOT/reports/log-statistics.txt"
    
    echo "Log Analysis Statistics" > "$stats_file"
    echo "Generated: $(date)" >> "$stats_file"
    echo "" >> "$stats_file"
    
    echo "=== Log Entry Counts ===" >> "$stats_file"
    for logfile in "$LOG_ANALYSIS_ROOT/logs"/*.log; do
        [ -f "$logfile" ] || continue
        echo "$(basename "$logfile"): $(wc -l < "$logfile") lines" >> "$stats_file"
    done
    
    echo "" >> "$stats_file"
    echo "Total errors: $(grep -i error "$LOG_ANALYSIS_ROOT/logs"/*.log 2>/dev/null | wc -l)" >> "$stats_file"
    echo "Total warnings: $(grep -i warning "$LOG_ANALYSIS_ROOT/logs"/*.log 2>/dev/null | wc -l)" >> "$stats_file"

    log_message "INFO" "Log statistics generated"
}

# dispatcher (required)
case "$1" in
    gather_system_logs) gather_system_logs "$2" ;;
    generate_log_statistics) generate_log_statistics ;;
    *) echo "Usage: $0 {gather_system_logs|generate_log_statistics}" ;;
esac
EOF

#!/bin/bash

LOG_ANALYSIS_ROOT="/opt/log-analysis"
source /opt/log-analysis/scripts/log-analyzer.sh

parse_security_events() {
    local logfile="$LOG_ANALYSIS_ROOT/logs/secure.log"
    local output_file="$LOG_ANALYSIS_ROOT/analysis/security/security-events.txt"

    log_message "INFO" "Parsing security events"

    {
        echo "Security Event Analysis"
        echo "Generated: $(date)"
        echo ""

        echo "=== Failed Login Attempts ==="
        grep -i "failed password" "$logfile" 2>/dev/null | head -20

        echo ""
        echo "=== Invalid Users ==="
        grep -i "invalid user" "$logfile" 2>/dev/null | head -20

        echo ""
        echo "=== Root Login Events ==="
        grep -i "accepted.*root" "$logfile" 2>/dev/null | head -10

        echo ""
        echo "=== Sudo Activities ==="
        grep -i "sudo" "$logfile" 2>/dev/null | head -20

    } > "$output_file"

    log_message "INFO" "Security parsing finished"
}

parse_system_errors() {
    local logfile="$LOG_ANALYSIS_ROOT/logs/messages.log"
    local output_file="$LOG_ANALYSIS_ROOT/analysis/errors/system-errors.txt"

    log_message "INFO" "Parsing system errors"

    {
        echo "System Error Analysis"
        echo "Generated: $(date)"
        echo ""

        echo "=== Critical Errors ==="
        grep -Ei "critical|fatal|panic" "$logfile" 2>/dev/null | head -20

        echo ""
        echo "=== Hardware Errors ==="
        grep -Ei "disk|memory|hardware" "$logfile" 2>/dev/null | grep -i error | head -20

        echo ""
        echo "=== Service Failures ==="
        grep -Ei "failed|failure" "$logfile" 2>/dev/null | grep -i service | head -20

    } > "$output_file"

    log_message "INFO" "System error parsing finished"
}

parse_performance_issues() {
    local logfile="$LOG_ANALYSIS_ROOT/logs/journal-7days.log"
    local output_file="$LOG_ANALYSIS_ROOT/analysis/performance/performance-issues.txt"

    log_message "INFO" "Parsing performance issues"

    {
        echo "Performance Analysis"
        echo "Generated: $(date)"
        echo ""

        echo "=== High Load ==="
        grep -i "load average" "$logfile" 2>/dev/null | head -10

        echo ""
        echo "=== Memory Issues ==="
        grep -Ei "out of memory|oom|killed process" "$logfile" 2>/dev/null | head -10

        echo ""
        echo "=== Disk Issues ==="
        grep -Ei "no space|disk full|i/o error" "$logfile" 2>/dev/null | head -10

    } > "$output_file"

#!/bin/bash

LOG_ANALYSIS_ROOT="/opt/log-analysis"
source /opt/log-analysis/scripts/log-analyzer.sh

# Load pattern definitions
load_patterns() {
    local pattern_file="$1"

    if [ -f "$pattern_file" ]; then
        source "$pattern_file"
        log_message "INFO" "Loaded patterns from: $pattern_file"
    else
        log_message "ERROR" "Pattern file not found: $pattern_file"
        return 1
    fi
}

# Search for security patterns
search_security_patterns() {
    local logfile="$LOG_ANALYSIS_ROOT/logs/secure.log"
    local output_file="$LOG_ANALYSIS_ROOT/analysis/security/pattern-matches.txt"

    log_message "INFO" "Searching for security patterns in $logfile"

    load_patterns "$LOG_ANALYSIS_ROOT/patterns/security-patterns.conf" || exit 1

    {
        echo "Security Pattern Analysis"
        echo "Generated: $(date)"
        echo "Source: $logfile"
        echo ""

        echo "=== Failed Login Pattern Matches ==="
        echo "Count: $(grep -E "$PATTERN_FAILED_LOGIN" "$logfile" 2>/dev/null | wc -l)"
        grep -E "$PATTERN_FAILED_LOGIN" "$logfile" 2>/dev/null | head -5

        echo ""
        echo "=== Invalid User Pattern Matches ==="
        echo "Count: $(grep -E "$PATTERN_INVALID_USER" "$logfile" 2>/dev/null | wc -l)"
        grep -E "$PATTERN_INVALID_USER" "$logfile" 2>/dev/null | head -5

        echo ""
        echo "=== Root Login Pattern Matches ==="
        echo "Count: $(grep -E "$PATTERN_ROOT_LOGIN" "$logfile" 2>/dev/null | wc -l)"
        grep -E "$PATTERN_ROOT_LOGIN" "$logfile" 2>/dev/null | head -5

    } > "$output_file"

    log_message "INFO" "Security pattern search completed"
}

# Custom pattern search
custom_pattern_search() {
    local logfile="$1"
    local pattern="$2"
    local description="$3"

    local outfile="$LOG_ANALYSIS_ROOT/analysis/custom-patterns.txt"

    log_message "INFO" "Searching custom pattern: $description"

    local matches=$(grep -c "$pattern" "$logfile" 2>/dev/null)

    {
        echo "Pattern: $description"
        echo "Matches: $matches"

        if [ "$matches" -gt 0 ]; then
            echo "Sample matches:"
            grep "$pattern" "$logfile" | head -3
        fi
        echo "---"
    } >> "$outfile"
}

case "$1" in
    search_security_patterns) search_security_patterns ;;
    custom_pattern_search) custom_pattern_search "$2" "$3" "$4" ;;
    *) echo "Usage: $0 {search_security_patterns|custom_pattern_search}" ;;
esac
EOF

#!/bin/bash

source /opt/log-analysis/scripts/core.sh

ALERT_FILE="$LOG_ANALYSIS_ROOT/reports/alerts.txt"
SECURE_LOG="$LOG_ANALYSIS_ROOT/logs/secure.log"
MESSAGE_LOG="$LOG_ANALYSIS_ROOT/logs/messages.log"

########################################
# Reset alert file every run
########################################
> "$ALERT_FILE"

########################################
# Check for critical events
########################################
check_critical_events() {

    log_message "INFO" "Running alert detection"

    local alerts_found=0

    failed_logins=$(grep -c "Failed password" "$SECURE_LOG" 2>/dev/null)
    if (( failed_logins > 20 )); then
        echo "SECURITY ALERT: $failed_logins failed login attempts" >> "$ALERT_FILE"
        ((alerts_found++))
    fi

    root_logins=$(grep -c "session opened for user root" "$SECURE_LOG" 2>/dev/null)
    if (( root_logins > 5 )); then
        echo "PRIVILEGE ALERT: Excessive root sessions ($root_logins)" >> "$ALERT_FILE"
        ((alerts_found++))
    fi

    oom_events=$(grep -c "Out of memory" "$MESSAGE_LOG" 2>/dev/null)
    if (( oom_events > 0 )); then
        echo "SYSTEM ALERT: OOM Killer triggered ($oom_events)" >> "$ALERT_FILE"
        ((alerts_found++))
    fi

    disk_errors=$(grep -c "I/O error" "$MESSAGE_LOG" 2>/dev/null)
    if (( disk_errors > 0 )); then
        echo "DISK ALERT: Disk I/O errors detected ($disk_errors)" >> "$ALERT_FILE"
        ((alerts_found++))
    fi

    if (( alerts_found == 0 )); then
        echo "No critical alerts detected" >> "$ALERT_FILE"
    fi

    log_message "INFO" "Alert scan finished ($alerts_found alerts)"
}

########################################
# realtime monitor (manual only)
########################################
realtime_monitor() {
    local logfile="${1:-/var/log/messages}"

    log_message "INFO" "Realtime monitoring started: $logfile"

    tail -Fn0 "$logfile" | while read -r line; do
        if echo "$line" | grep -qiE "error|fail|critical"; then
            log_message "ALERT" "$line"
        fi
    done
}

########################################
case "$1" in
    scan) check_critical_events ;;
    live) realtime_monitor "$2" ;;
    *) echo "Usage: $0 {scan|live}" ;;
esac
EOF
